━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  WHAT YOU NEED:
  ─────────────
  ✅ Windows 10 or 11
  ✅ NVIDIA GPU (GTX 1060 or better)
  ✅ ~10 GB free disk space
  ✅ Internet (only for first-time setup)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  HOW TO INSTALL (ONE STEP):
  ──────────────────────────
  1. Extract this ZIP folder anywhere on your PC
  2. Open the extracted folder
  3. RIGHT-CLICK on "Install_JARVIS.bat" → "Run as administrator"
  4. Follow the on-screen instructions
  5. Wait for everything to install (~10-15 minutes)
  6. Done! A "Start_JARVIS.bat" shortcut will appear on your Desktop

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  HOW TO USE JARVIS:
  ──────────────────
  1. Double-click "Start_JARVIS.bat" on your Desktop
  2. Wait ~15-30 seconds for models to load
  3. Open http://127.0.0.1:8000 in your browser (Chrome/Brave/Edge)
  4. Show your hand 🖐️ to the camera for authentication
  5. Start talking to JARVIS!

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  VOICE COMMANDS:
  ───────────────
  🎤 "Open Chrome"         → Opens Google Chrome
  🎤 "Open Spotify"        → Opens Spotify
  🎤 "What time is it?"    → Tells the current time
  🎤 "Calculate 25 × 4"    → Does math
  🎤 "Search for Python"   → Searches the web
  🎤 "Tell me a joke"      → JARVIS tells a joke

  Or just talk to JARVIS about anything!

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  TROUBLESHOOTING:
  ────────────────
  ❌ "Site can't be reached"
     → Make sure you're using http:// (not https://)
     → Try http://127.0.0.1:8000 instead of localhost

  ❌ "Mic not working"
     → Click the 🔒 icon next to URL bar
     → Allow Microphone permissions

  ❌ "Camera not working"
     → Click the 🔒 icon next to URL bar
     → Allow Camera permissions
